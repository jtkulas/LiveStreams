---
title: "Awesome Slidez"
format: revealjs
theme: yourlook.scss
---
/*-- scss:defaults --*/        

@import url('https://fonts.googleapis.com/css?family=Irish Grover');   
@import url('https://fonts.googleapis.com/css?family=UnifrakturMaguntia');  

// Include theme-specific fonts
$systemFontsSansSerif: UnifrakturMaguntia, Irish Grover, arial;
$systemFontsMono: UnifrakturMaguntia, Irish Grover, arial;

$background: #353E43;
$foreground: #ffffe4;
$selection: #44475a;
$comment: #6272a4;
$red: #ff5555;
$orange: #ffb86c;
$yellow: #f1fa8c;
$green: #50fa7b;
$purple: #9D00FF;
$cyan: #8be9fd;
$pink: #ff79c6;
$badass: #bada55;

// Override theme settings
$body-color: $foreground !default;
$presentation-heading-color: $cyan !default;
$presentation-font-size-root: 60px !default;
$presentation-heading-text-shadow: none !default;
$presentation-heading-text-transform: none !default;
$body-bg: $background !default;
$link-color: $pink !default;
$link-color-hover: $badass !default;
$selection-bg: $selection !default;
$code-color: $green !default;
$presentation-list-bullet-color: $foreground !default;

$font-family-sans-serif: $systemFontsSansSerif !default;
$font-family-monospace: "Fira Code", $systemFontsMono !default;

// Change text colors against light slide backgrounds
$light-bg-text-color: $body-bg !default;

/*-- scss:rules --*/

// Define additional color effects based on Dracula spec
// https://spec.draculatheme.com/
:root {
  --r-bold-color: #{$badass};
  --r-italic-color: #{$yellow};
  --r-inline-code-color: #{$code-color};
  --r-list-bullet-color: #{$presentation-list-bullet-color};
}

.reveal {
  strong,
  b {
    &:not(.callout-title strong, .callout-tile b) {
      color: var(--r-bold-color);
    }
  }
  em,
  i,
  blockquote {
    color: var(--r-italic-color);
  }
  code {
    color: var(--r-inline-code-color);
  }
  // Dracula colored list bullets and numbers
  ul,
  ol {
    li::marker {
      color: var(--r-list-bullet-color);
    }
  }
}

/**
 * Dracula colors by Zeno Rocha
 * https://draculatheme.com/contribute
 */
html * {
  color-profile: sRGB;
  rendering-intent: auto;
}
  1. Sequence of scss:defaults vs scss:rules important for Quarto docs
  2. Easy way to grab fonts – requires internet connection
